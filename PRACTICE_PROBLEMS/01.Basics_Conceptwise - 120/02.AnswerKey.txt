         ROW_NUMBER() OVER (
           PARTITION BY department_id
           ORDER BY salary DESC

4.
SELECT *
FROM (

         DENSE_RANK() OVER (
           PARTITION BY department_id
           ORDER BY salary DESC
         ) AS rnk
  FROM Employee e

5.
SELECT *
FROM (
  SELECT e.*,
         ROW_NUMBER() OVER (
           PARTITION BY department_id
           ORDER BY salary DESC
         ) AS rn
  FROM Employee e

6.
SELECT name, salary,
       ROW_NUMBER() OVER (ORDER BY salary DESC) AS row_num,
       RANK() OVER (ORDER BY salary DESC) AS rank_num,
       DENSE_RANK() OVER (ORDER BY salary DESC) AS dense_rank_num

7.
SELECT employee_id, sale_date, amount,
       LAG(amount) OVER (
         PARTITION BY employee_id
         ORDER BY sale_date
       ) AS previous_amount

8.
SELECT employee_id, sale_date, amount,
       LAG(amount) OVER (
         PARTITION BY employee_id
         ORDER BY sale_date
       ) AS previous_amount,
       amount -
       LAG(amount) OVER (
         PARTITION BY employee_id
         ORDER BY sale_date
       ) AS difference
FROM Sales;

9.

  SELECT employee_id, sale_date, amount,
         LAG(amount) OVER (
           PARTITION BY employee_id
           ORDER BY sale_date
         ) AS previous_amount
  FROM Sales

WHERE amount > previous_amount;
